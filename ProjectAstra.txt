do
    local getgc_   = getfenv().getgc or getgc
    local hookfunc = getfenv().hookfunction or hookfunction
    local newcc    = getfenv().newcclosure or function(f) return f end
    local hookmm   = getfenv().hookmetamethod or hookmetamethod
    local getncm   = getfenv().getnamecallmethod or getnamecallmethod

    local R = {}
    local function isGuarded(inst)
        if R[inst] then return true end
        local n = inst.Name
        if #n == 36 and string.match(
            n,
            "^%x%x%x%x%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%x%x%x%x%x%x%x%x$"
        ) then
            R[inst] = true
            return true
        end
        return false
    end

    local origFire
    origFire = hookfunc(Instance.new("RemoteEvent").FireServer, function(self, ...)
        if isGuarded(self) then return origFire(self, ...) end
        for _, v in ipairs({...}) do
            if type(v) == "string" and string.find(string.lower(v), "kick") then
                return
            end
        end
        return origFire(self, ...)
    end)

    local origNC
    origNC = hookmm(game, "__namecall", newcc(function(self, ...)
        local m = getncm()
        if self == LocalPlayer and (m == "Kick" or m == "kick" or m == "Destroy" or m == "destroy") then
            return
        end
        return origNC(self, ...)
    end))

    local origKick
    origKick = hookfunc(LocalPlayer.Kick, newcc(function(self, ...)
        if not checkcaller() and self == LocalPlayer then return end
        return origKick(self, ...)
    end))

    local origDestroy
    origDestroy = hookfunc(LocalPlayer.Destroy, newcc(function(self, ...)
        if not checkcaller() and self == LocalPlayer then return end
        return origDestroy(self, ...)
    end))

    pcall(function()
        for _, v in pairs(getgc_(true)) do
            if type(v) == "table" then
                pcall(function()
                    if rawget(v, "Send") and type(rawget(v, "Send")) == "function"
                        and rawget(v, "Get") and rawget(v, "Encrypt") then
                        local origSend
                        origSend = hookfunc(v.Send, newcc(function(cmd, ...)
                            if type(cmd) == "string" then
                                local c = string.lower(cmd)
                                if c == "detected" or c == "logerror" then return end
                            end
                            return origSend(cmd, ...)
                        end))
                    end
                    if rawget(v, "Kill") and type(rawget(v, "Kill")) == "function"
                        and rawget(v, "Disconnect") then
                        hookfunc(v.Kill, newcc(function(...) return end))
                        hookfunc(v.Disconnect, newcc(function(...) return end))
                    end
                end)
            end
        end
    end)
end

local Essentials = {}
do
    local function getLocalHRP()
        local char = LocalPlayer.Character
        return char and char:FindFirstChild("HumanoidRootPart")
    end

    function Essentials.getLowestHealthNearbyPlayer()
        local localHRP = getLocalHRP()
        if not localHRP then return nil end

        local lowestHealth = math.huge
        local targetPlayer = nil

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
                local targetHRP = player.Character:FindFirstChild("HumanoidRootPart")
                if humanoid and targetHRP and humanoid.Health > 0 then
                    local distance = (targetHRP.Position - localHRP.Position).Magnitude
                    if distance <= _G.strafeRange and humanoid.Health < lowestHealth then
                        lowestHealth = humanoid.Health
                        targetPlayer = player
                    end
                end
            end
        end

        return targetPlayer
    end

    function Essentials.getClosestPlayer()
        local hrp = getLocalHRP()
        if not hrp then return nil end

        local closest = nil
        local shortest = math.huge

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local targetHRP = player.Character:FindFirstChild("HumanoidRootPart")
                if targetHRP then
                    local dist = (hrp.Position - targetHRP.Position).Magnitude
                    if dist < shortest then
                        shortest = dist
                        closest = player
                    end
                end
            end
        end

        return closest
    end
end

local KillAura = {}
do
    local running = false
    local attackRemote

    local function getAttack()
        if attackRemote then return attackRemote end
        local remotes = RS:WaitForChild("GameRemotes", 10)
        if not remotes then return nil end
        attackRemote = remotes:WaitForChild("Attack", 10)
        return attackRemote
    end

    local function loop()
        while running do
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local attack = getAttack()

            if hrp and attack then
                local target
                if _G.selectedTargeting == "lowest" then
                    target = Essentials.getLowestHealthNearbyPlayer()
                else
                    target = Essentials.getClosestPlayer()
                end

                if target and target.Character then
                    local tHRP = target.Character:FindFirstChild("HumanoidRootPart")
                    if tHRP then
                        local d = hrp.Position - tHRP.Position
                        if (d.X * d.X + d.Z * d.Z) <= _G.RANGE_SQ then
                            pcall(function()
                                attack:InvokeServer(target.Character)
                            end)
                        end
                    end
                end
            end

            task.wait(_G.delay or 0.1)
        end
    end

    function KillAura.start()
        if running then return end
        running = true
        task.spawn(loop)
    end

    function KillAura.stop()
        running = false
    end
end

local TargetStrafe = {}
do
    local running = false
    local hbConn
    local timeAcc = 0

    local function onHeartbeat(dt)
        if not running then return end

        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local target
        if _G.selectedTargeting == "lowest" then
            target = Essentials.getLowestHealthNearbyPlayer()
        else
            target = Essentials.getClosestPlayer()
        end

        local tChar = target and target.Character
        local tHRP = tChar and tChar:FindFirstChild("HumanoidRootPart")
        if not tHRP then return end

        timeAcc = timeAcc + dt * _G.speed

        local offset = Vector3.new(
            math.cos(timeAcc) * _G.radius,
            0,
            math.sin(timeAcc) * _G.radius
        )

        local targetPos = tHRP.Position
        hrp.CFrame = CFrame.new(targetPos + offset, targetPos)
    end

    function TargetStrafe.start()
        if running then return end
        running = true
        timeAcc = 0
        hbConn = RunService.Heartbeat:Connect(onHeartbeat)
    end

    function TargetStrafe.stop()
        running = false
        if hbConn then
            hbConn:Disconnect()
            hbConn = nil
        end
    end
end

local Triggerbot = {}
do
    local running = false
    local currentTarget = nil
    local RANGE_SQ = 20 * 20

    local function step()
        if not running then
            currentTarget = nil
            return
        end

        local char = LocalPlayer.Character
        if not char then return end

        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local cam = Workspace.CurrentCamera
        if not cam then return end

        local viewport = cam.ViewportSize
        local ray = cam:ViewportPointToRay(viewport.X / 2, viewport.Y / 2)

        local params = RaycastParams.new()
        params.FilterDescendantsInstances = { char }
        params.FilterType = Enum.RaycastFilterType.Blacklist

        local result = Workspace:Raycast(ray.Origin, ray.Direction * 500, params)

        local newTarget = nil
        if result then
            local model = result.Instance:FindFirstAncestorOfClass("Model")
            if model and model ~= char and model:FindFirstChildOfClass("Humanoid") then
                newTarget = model
            end
        end

        currentTarget = newTarget
        if not currentTarget then return end

        local tHRP = currentTarget:FindFirstChild("HumanoidRootPart")
        local hum = currentTarget:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 or not tHRP then return end

        local d = hrp.Position - tHRP.Position
        if (d.X * d.X + d.Z * d.Z) <= RANGE_SQ then
            local attack = RS:WaitForChild("GameRemotes"):WaitForChild("Attack")
            pcall(function()
                attack:InvokeServer(currentTarget)
            end)
        end
    end

    local function loop()
        while running do
            step()
            task.wait()
        end
    end

    function Triggerbot.start()
        if running then return end
        running = true
        task.spawn(loop)
    end

    function Triggerbot.stop()
        running = false
        currentTarget = nil
    end
end

local Jesus = {}

        return part:IsA("BasePart")
            and (part.Name == "Water" or part.Name == "Lava")
    end

    local function setCollision(state)
        if not fluidFolder then return end

        for _, obj in ipairs(fluidFolder:GetDescendants()) do
            if isWater(obj) then
                obj.CanCollide = state
            end
        end
    end

    function Jesus.start()
        if enabled then return end
        enabled = true
        if not fluidFolder then return end

        setCollision(true)

        conn = fluidFolder.DescendantAdded:Connect(function(obj)

        enabled = false

        if conn then
            conn:Disconnect()
            conn = nil
        end

        setCollision(false)
    end

local AutoArmor = {}
do
    local running = false
    local moveItem
    local itemInfo

    local TIERS = { Diamond = 4, Ruby = 3, Iron = 2, Gold = 2, Steel = 2, Stone = 1 }
    local ARMOR = { Helmet = 103, Chestplate = 102, Leggings = 101, Boots = 100 }

    local function getMoveItem()
        if moveItem then return moveItem end
        local remotes = RS:WaitForChild("GameRemotes", 10)
        if not remotes then return nil end
        moveItem = remotes:WaitForChild("MoveItem", 10)
        return moveItem
    end

    local function getItemInfo()
        if itemInfo then return itemInfo end
        local mod = RS:WaitForChild("AssetsMod", 10)
        if not mod then return nil end
        local info = mod:WaitForChild("ItemInfo", 10)
        if not info then return nil end
        local ok, result = pcall(require, info)
        if ok then itemInfo = result end
        return itemInfo
    end

    local function decode(slot)
        local ok, data = pcall(function()
            return Http:JSONDecode(slot.Value)
        end)
        return ok and data or nil
    end

    local function getTier(name)
        for prefix, tier in pairs(TIERS) do
            if string.find(name, "^" .. prefix) then
                local afterPrefix = string.sub(name, #prefix + 1, #prefix + 1)
                if afterPrefix == "" or afterPrefix == " " or string.match(afterPrefix, "%u") then
                    return tier
                end
            end
        end
        local info = getItemInfo()
        local entry = info and info[name]
        return entry and (entry.tier or entry.Tier) or 0
    end

    local function getArmorType(name)
        for aType, id in pairs(ARMOR) do
            if string.find(name, aType) then
                return id
            end
        end
        return nil
    end

    local function isBetter(new, old)
        local newTier = getTier(new.name)
        local oldTier = getTier(old.name)

        local newDur = new.durability or 0
        local oldDur = old.durability or 0

        local info = getItemInfo()
        local newInfo = info and info[new.name]
        local oldInfo = info and info[old.name]
        local newMax = (newInfo and newInfo.durability) or 100
        local oldMax = (oldInfo and oldInfo.durability) or 100

        if oldDur <= (oldMax * 0.1) and newDur > (newMax * 0.1) then
            if newTier >= (oldTier - 1) then
                return true
            end
        end

        if newTier > oldTier then
            if newDur > (newMax * 0.1) or oldDur <= (oldMax * 0.05) then
                return true
            end
        elseif newTier == oldTier then
            if newDur > oldDur then
                return true
            end
        end

        if newTier == oldTier and newDur == oldDur then
            return newMax > oldMax
        end

        return false
    end

    local function autoEquipArmor()
        local move = getMoveItem()
        if not move then return end

        local char = LocalPlayer.Character
        if not char then return end

        local inv = char:FindFirstChild("Inventory")
        if not inv then return end

        local best = {
            [103] = { tier = -1, dur = -1, idx = nil },
            [102] = { tier = -1, dur = -1, idx = nil },
            [101] = { tier = -1, dur = -1, idx = nil },
            [100] = { tier = -1, dur = -1, idx = nil },
        }

        for i = 0, 35 do
            local slot = inv:FindFirstChild("Slot" .. i)
            local data = slot and decode(slot)

            if data and data.count and data.count >= 1 and data.name then
                local armorSlot = getArmorType(data.name)
                if armorSlot then
                    local tier = getTier(data.name)
                    local dur = data.durability or 0

                    if tier > best[armorSlot].tier
                        or (tier == best[armorSlot].tier and dur > best[armorSlot].dur) then
                        best[armorSlot] = { tier = tier, dur = dur, idx = i, name = data.name }
                    end
                end
            end
        end

        for slot, item in pairs(best) do
            if item.idx then
                local equippedSlot = inv:FindFirstChild("Slot" .. slot)
                local equipped = equippedSlot and decode(equippedSlot)

                if not equipped or (equipped.count or 0) <= 0 then
                    pcall(function() move:InvokeServer(item.idx, slot, true) end)
                elseif isBetter(item, equipped) then
                    pcall(function() move:InvokeServer(item.idx, slot, true) end)
                end
            end
        end
    end

    local function loop()
        while running do
            autoEquipArmor()
            task.wait()
        end
    end

    function AutoArmor.start()
        if running then return end
        running = true
        task.spawn(loop)
    end

    function AutoArmor.stop()
        running = false
    end
end

local AutoTool = {}
do
    local running = false
    local CGlobals, BlockInfo, ItemInfo, ItemLevels, BlockHighlights

    local function getModules()
        if CGlobals and BlockInfo and ItemInfo and ItemLevels and BlockHighlights then return true end

        local char = LocalPlayer.Character
        if not char then return false end
        local inv = char:FindFirstChild("Inventory")
        if not inv then return false end

        local ps = LocalPlayer:FindFirstChild("PlayerScripts")
        if not ps then return false end
        local main = ps:FindFirstChild("MainLocalScript")
        if not main then return false end

        local assetsMod = RS:FindFirstChild("AssetsMod")
        if not assetsMod then return false end

        local cg = main:FindFirstChild("CGlobals")
        if cg then
            local ok, res = pcall(require, cg)
            if ok then CGlobals = res end
        end

        local bh = main:FindFirstChild("BlockHighlights")
        if bh then
            local ok, res = pcall(require, bh)
            if ok then BlockHighlights = res end
        end

        local bi = assetsMod:FindFirstChild("BlockInfo")
        if bi then
            local ok, res = pcall(require, bi)
            if ok then BlockInfo = res end
        end

        local ii = assetsMod:FindFirstChild("ItemInfo")
        if ii then
            local ok, res = pcall(require, ii)
            if ok then ItemInfo = res end
        end

        local il = assetsMod:FindFirstChild("ItemLevels")
        if il then
            local ok, res = pcall(require, il)
            if ok then ItemLevels = res end
        end

        return CGlobals ~= nil
            and BlockInfo ~= nil
            and ItemInfo ~= nil
            and ItemLevels ~= nil
            and BlockHighlights ~= nil
    end

    local function getHotbar()
        local hud = pg:FindFirstChild("HUDGui")
        return hud and hud:FindFirstChild("Hotbar")
    end

    local function getSlotButton(index)
        local hotbar = getHotbar()
        if not hotbar then return nil end
        local targetX = index * 40 + 6
        for _, btn in ipairs(hotbar:GetChildren()) do
            if btn:IsA("TextButton") and btn.Position.X.Offset == targetX then
                return btn
            end
        end
        return nil
    end

    local function setSlot(index)
        local slot = getSlotButton(index)
        if not slot then return end
        for _, conn in ipairs(getconnections(slot.MouseButton1Click)) do
            conn:Fire()
        end
    end

    local function decodeSlot(slotVal)
        local ok, data = pcall(function()
            return Http:JSONDecode(slotVal.Value)
        end)
        if ok and data and data.count and data.count > 0 then
            return data
        end
        return nil
    end

    local function getItemSpeed(itemName, reqType, betterTool)
        local itemData = ItemInfo and ItemInfo[itemName]
        if itemData and (itemData.tooltype == reqType or itemData.tooltype == betterTool) then
            return (ItemLevels and ItemLevels.speedMul[itemData.level]) or 1
        end
        return 1
    end

    local function getBestToolSlot(blockName)
        local blockData = BlockInfo and BlockInfo[blockName]
        if not blockData then return nil end

        local char = LocalPlayer.Character
        if not char then return nil end
        local inv = char:FindFirstChild("Inventory")
        if not inv then return nil end

        local bestSlot, maxSpeed = nil, 0
        local reqType, betterTool = blockData.toolRequire, blockData.betterTool

        for i = 0, 8 do
            local slotVal = inv:FindFirstChild("Slot" .. i)
            if slotVal then
                local data = decodeSlot(slotVal)
                if data then
                    local speed = getItemSpeed(data.name, reqType, betterTool)
                    if speed > maxSpeed then
                        maxSpeed = speed
                        bestSlot = i
                    end
                end
            end
        end

        return bestSlot
    end

    local function getSelectedSlotObj()
        local charModel = Workspace:FindFirstChild(LocalPlayer.Name)
        return charModel and charModel:FindFirstChild("SelectedSlot")
    end

    local function loop()
        while running do
            if getModules() then
                local selectedSlotObj = getSelectedSlotObj()
                if selectedSlotObj then
                    local targetPos = CGlobals.TargetBlockCoordinate
                    local targetBlock = CGlobals.BlockUnderMouse

                    if targetPos and targetBlock and BlockHighlights.IsBreaking(targetPos) then
                        local bestSlot = getBestToolSlot(targetBlock.Name)
                        if bestSlot and selectedSlotObj.Value ~= bestSlot then
                            pcall(setSlot, bestSlot)
                        end
                    end
                end
            end
            task.wait()
        end
    end

    function AutoTool.start()

local SkyBuild = {}
do
    local running = false
    local coordConn
    local M_World, BlocksByName

    local function loadModules()
        if M_World and BlocksByName then return true end
        local ok1, world = pcall(function()
            return require(LocalPlayer.PlayerScripts.MainLocalScript.CWorld)
        end)
        local ok2, ids = pcall(function()
            return require(RS.AssetsMod.IDs)
        end)
        if ok1 then M_World = world end
        if ok2 and ids and ids.ByName and ids.ByName.Blocks then
            BlocksByName = ids.ByName.Blocks
        end
        return M_World ~= nil and BlocksByName ~= nil
    end

    local function onCoordChanged()
        if not running then return end
        if not loadModules() then return end

        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end

        local selected = char:FindFirstChild("SelectedSlot")
        local placeSlot = selected and selected.Value
        if not placeSlot then return end

        local hud = pg:FindFirstChild("HUDGui")
        local inv = hud and hud:FindFirstChild("Inventory")
        local slots = inv and inv:FindFirstChild("Slots")
        local slotGui = slots and slots:FindFirstChild("Slot" .. placeSlot)
        if not slotGui or not slotGui.Slot.Display:FindFirstChild("SlotB") then return end

        local realBlock
        for _, v in pairs(slotGui.Slot.Display.SlotB:GetChildren()) do
            realBlock = v.Name
            break
        end
        if not realBlock then return end

        local itemblock_info = BlocksByName[realBlock]
        if not itemblock_info then return end

        local df = hud:FindFirstChild("DataFrame")
        local coordText = df and df:FindFirstChild("Coord")
        if not coordText then return end

        local x, y, z = coordText.Text:match("(%-?%d+),%s*(%-?%d+),%s*(%-?%d+)")
        if not x then return end

        x = tonumber(x)
        y = tonumber(y) - 1
        z = tonumber(z)

        local block, chunk = M_World.getBlock(x, y, z)
        local canPlace = false
        if not block then
            canPlace = true
        else
            for _, v in pairs(block) do
                if v == 0 then
                    canPlace = true
                    break
                end
            end
        end
        if not canPlace then return end

        M_World.placeBlock(x, y, z, chunk, 1, itemblock_info.id)
        local ok, name = RS.GameRemotes.PlaceBlock:InvokeServer(x, y, z, placeSlot, 1)
        if not ok then
            chunk:change(x % 16, y, z % 16, name)
        end
    end

    function SkyBuild.start()
        if running then return end
        running = true
        local hud = pg:FindFirstChild("HUDGui")
        local df = hud and hud:FindFirstChild("DataFrame")
        local coordText = df and df:FindFirstChild("Coord")
        if coordText then
            coordConn = coordText:GetPropertyChangedSignal("Text"):Connect(onCoordChanged)
        end
    end

    function SkyBuild.stop()
        if not running then return end
        running = false
        if coordConn then
            coordConn:Disconnect()
            coordConn = nil
        end
    end
end

local ReloadChunks = {}
do
    function ReloadChunks.start()
        local ps = LocalPlayer:FindFirstChild("PlayerScripts")
        local main = ps and ps:FindFirstChild("MainLocalScript")
        local cworldObj = main and main:FindFirstChild("CWorld")
        if not cworldObj then return end

        local ok, CWorld = pcall(require, cworldObj)
        if not ok or type(CWorld) ~= "table" then return end

        local world = CWorld.World
        local loadingChunks = CWorld.LoadingChunks
        local renderingChunks = CWorld.RenderingChunks
        local loadingReqQueue = CWorld.LoadingReqQueue
        local processingBlocks = CWorld.ProcessingBlocks

        if renderingChunks then
            for _, yrow in pairs(renderingChunks) do
                for _, chunk in pairs(yrow) do
                    if chunk.BlockerPart then
                        pcall(function() chunk.BlockerPart:Destroy() end)
                    end
                    if chunk.vfold then
                        pcall(function() chunk.vfold:Destroy() end)
                    end
                    if chunk.vlfold then
                        pcall(function() chunk.vlfold:Destroy() end)
                    end
                end
            end
        end

        if world then table.clear(world) end
        if loadingChunks then table.clear(loadingChunks) end
        if renderingChunks then table.clear(renderingChunks) end
        if loadingReqQueue then table.clear(loadingReqQueue) end

        if processingBlocks and typeof(processingBlocks) == "table" then
            if processingBlocks.clear then
                pcall(function() processingBlocks:clear() end)
            elseif processingBlocks.Contents then
                table.clear(processingBlocks.Contents)
            else
                table.clear(processingBlocks)
            end
        end
    end

    function ReloadChunks.stop() end
end

local NameProtect = {}
do
    local running = false
    local conns = {}
    local savedName
    local PROTECT_VAL = "Protected"

    local function isTextLabel(v)
        return v.ClassName == "TextLabel"
    end

    local function applyTo(label)
        if not running then return end
        if string.find(label.Text, savedName, 1, true) then
            label.Text = label.Text:gsub(savedName, PROTECT_VAL)
        end
    end

    local function hookLabel(label)
        applyTo(label)
        table.insert(conns, label:GetPropertyChangedSignal("Text"):Connect(function()
            applyTo(label)
        end))
    end

    function NameProtect.start()
        if running then return end
        running = true
        savedName = LocalPlayer.Name
        getgenv().name = PROTECT_VAL

        for _, v in next, game:GetDescendants() do
            if isTextLabel(v) then
                hookLabel(v)
            end
        end

        table.insert(conns, game.DescendantAdded:Connect(function(v)
            if running and isTextLabel(v) then
                hookLabel(v)
            end
        end))
    end

    function NameProtect.stop()
        if not running then return end
        running = false

        for _, c in ipairs(conns) do
            pcall(function() c:Disconnect() end)
        end
        table.clear(conns)
    end
end

local AirWalk = {}
do
    local running = false
    local platform
    local loopThread

    local function makePlatform()
        local p = Instance.new("Part")
        p.Anchored = true
        p.Size = Vector3.new(5, 1, 5)
        p.Transparency = 1
        p.CanCollide = true
        p.Parent = Workspace
        return p
    end

    local function loop()
        local platformY
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            platformY = hrp.Position.Y - 3
        end

        while running do
            local c = LocalPlayer.Character
            local h = c and c:FindFirstChild("HumanoidRootPart")
            if h and platform then
                if not platformY then
                    platformY = h.Position.Y - 3
                end
                platform.Position = Vector3.new(h.Position.X, platformY, h.Position.Z)
            end
            task.wait()
        end
    end

    function AirWalk.start()
        if running then return end
        running = true

        platform = makePlatform()

        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            platform.Position = Vector3.new(hrp.Position.X, hrp.Position.Y - 3, hrp.Position.Z)
        end

        loopThread = task.spawn(loop)
    end

    function AirWalk.stop()
        if not running then return end
        running = false

        if platform then
            platform:Destroy()
            platform = nil
        end
    end
end

local WallBuilderX = {}
do
    local running = false
    local coordConn
    local M_World, BlocksByName

    local function loadModules()
        if M_World and BlocksByName then return true end
        local ok1, world = pcall(function()

        end)

        if ok2 and ids and ids.ByName and ids.ByName.Blocks then
            BlocksByName = ids.ByName.Blocks
        end
        return M_World ~= nil and BlocksByName ~= nil
    end

    local function onCoordChanged()
        if not running then return end
        if not loadModules() then return end

        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end

        local selected = char:FindFirstChild("SelectedSlot")
        local placeSlot = selected and selected.Value
        if not placeSlot then return end

        local hud = pg:FindFirstChild("HUDGui")
        local inv = hud and hud:FindFirstChild("Inventory")
        local slots = inv and inv:FindFirstChild("Slots")
        local slotGui = slots and slots:FindFirstChild("Slot" .. placeSlot)
        if not slotGui or not slotGui.Slot.Display:FindFirstChild("SlotB") then return end

        local realBlock
        for _, v in pairs(slotGui.Slot.Display.SlotB:GetChildren()) do
            realBlock = v.Name
            break
        end
        if not realBlock then return end

        local itemblock_info = BlocksByName[realBlock]
        if not itemblock_info then return end

        local df = hud:FindFirstChild("DataFrame")
        local coordText = df and df:FindFirstChild("Coord")
        if not coordText then return end

        local x, y, z = coordText.Text:match("(%-?%d+),%s*(%-?%d+),%s*(%-?%d+)")
        if not x then return end

        x = tonumber(x)
        y = tonumber(y) - 1
        z = tonumber(z)

        for ox = -2, 2 do
            task.spawn(function()
                local px, py, pz = x + ox, y, z

                local block, chunk = M_World.getBlock(px, py, pz)
                local canPlace = false
                if not block then
                    canPlace = true
                else
                    for _, v in pairs(block) do
                        if v == 0 then
                            canPlace = true
                            break
                        end
                    end
                end
                if not canPlace then return end

                M_World.placeBlock(px, py, pz, chunk, 1, itemblock_info.id)
                local ok, name = RS.GameRemotes.PlaceBlock:InvokeServer(px, py, pz, placeSlot, 1)
                if not ok then
                    chunk:change(px % 16, py, pz % 16, name)
                end
            end)
        end
    end

    function WallBuilderX.start()
        if running then return end
        running = true
        local hud = pg:FindFirstChild("HUDGui")
        local df = hud and hud:FindFirstChild("DataFrame")
        local coordText = df and df:FindFirstChild("Coord")
        if coordText then
            coordConn = coordText:GetPropertyChangedSignal("Text"):Connect(onCoordChanged)
        end
    end

    function WallBuilderX.stop()
        if not running then return end
        running = false
        if coordConn then
            coordConn:Disconnect()
            coordConn = nil
        end
    end
end

            local slotObj = chr.Inventory:WaitForChild("Slot"..slot, 5)

            local slotStr = slotObj.Value
            local removed = string.gsub(slotStr, '[%{%}%"]', "")
            local parts = string.split(removed, ",")
            local name, count, durab
            for _, data in pairs(parts) do
                if data:match("name:") then
                    name = string.sub(data, 6)
                end
                if data:match("count:") then
                    count = string.sub(data, 7)
                end
                if data:match("durability:") then
                    durab = string.sub(data, 12)
                end
            end
            return name, count, durab
        else
            return false
        end

                    slots.dur.Visible = false

    local function hookGui()
        MIV.CB.MouseButton1Click:Connect(function()
            if MIV.CB.Text == "-" then
                MIV.BHIs.Visible = false
                MIV.CB.Text = "+"
            elseif MIV.CB.Text == "+" then
                MIV.BHIs.Visible = true
                MIV.CB.Text = "-"
            end
        end)

        MIV.PNs.MouseButton1Click:Connect(function()
            local found = userFromBox(MIV.PN.Text)
            if found ~= nil then
                local player = Players:FindFirstChild(tostring(found))
                if player ~= nil and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character:FindFirstChild("Humanoid").Health > 0 then
                    MIV.PN.PlaceholderText = "Loaded Inventory: "..player.Name
                    MIV.PN.Text = ""
                    local chr = player.Character
                    disconnectAll()
                    slotDataAction("ClearAll")
                    for i = -1, 35 do
                        local name, count, durab = getDataSlot(chr, i)
                        if name ~= nil and count ~= nil and tonumber(count) > 0 then
                            slotDataAction("UpdateSlot", chr, i, name, count, durab)
                        end
                        local slotObj = chr.Inventory:FindFirstChild("Slot"..i)
                        if slotObj then
                            table.insert(slotChangeEvents, slotObj:GetPropertyChangedSignal("Value"):Connect(function()
                                local n, c, d = getDataSlot(chr, i)
                                if n ~= nil and c ~= nil and tonumber(c) > 0 then
                                    slotDataAction("UpdateSlot", chr, i, n, c, d)
                                else
                                    slotDataAction("ClearSlot", nil, i)
                                end
                            end))
                        end
                    end
                    for i = 100, 103 do
                        local name, count, durab = getDataSlot(chr, i)
                        if name ~= nil and count ~= nil and tonumber(count) > 0 then
                            slotDataAction("UpdateSlot", chr, i, name, count, durab)
                        end
                        local slotObj = chr.Inventory:FindFirstChild("Slot"..i)
                        if slotObj then
                            table.insert(slotChangeEvents, slotObj:GetPropertyChangedSignal("Value"):Connect(function()
                                local n, c, d = getDataSlot(chr, i)
                                if n ~= nil and c ~= nil and tonumber(c) > 0 then
                                    slotDataAction("UpdateSlot", chr, i, n, c, d)
                                else
                                    slotDataAction("ClearSlot", nil, i)
                                end
                            end))
                        end
                    end
                    if chr:FindFirstChild("SelectedSlot") and MIV.IBrgI:FindFirstChild("Slot"..chr.SelectedSlot.Value) then
                        SOT.Parent = MIV.IBrgI:FindFirstChild("Slot"..chr.SelectedSlot.Value)
                        SOT.Enabled = true
                        table.insert(slotChangeEvents, chr.SelectedSlot:GetPropertyChangedSignal("Value"):Connect(function()
                            local result = chr.SelectedSlot.Value
                            if MIV.IBrgI:FindFirstChild("Slot"..result) then
                                SOT.Parent = MIV.IBrgI:FindFirstChild("Slot"..result)
                                SOT.Enabled = true
                            end
                        end))
                    end
                    table.insert(slotChangeEvents, Players.PlayerRemoving:Connect(function(plr)
                        if plr == player then
                            MIV.PN.Text = ""
                            MIV.PN.PlaceholderText = "Player Disconnected: "..plr.Name
                            disconnectAll()
                            slotDataAction("ClearAll")
                        end
                    end))
                    if chr:FindFirstChild("Humanoid") then
                        table.insert(slotChangeEvents, chr.Humanoid.Died:Connect(function()
                            MIV.PN.Text = ""
                            MIV.PN.PlaceholderText = "Player Died: "..player.Name
                            disconnectAll()
                            slotDataAction("ClearAll")
                        end))
                    end
                end
            else
                disconnectAll()
                slotDataAction("ClearAll")
                MIV.PNs.Interactable = false
                MIV.PN.Interactable = false
                MIV.PN.PlaceholderText = "Not Found"
                MIV.PN.PlaceholderColor3 = Color3.fromRGB(255, 0, 0)
                MIV.PN.Text = ""
                task.delay(1, function()
                    if MIV.PNs then MIV.PNs.Interactable = true end
                    if MIV.PN then
                        MIV.PN.Interactable = true
                        MIV.PN.PlaceholderText = "Player Here"
                        MIV.PN.PlaceholderColor3 = Color3.fromRGB(178, 178, 178)
                    end
                end)
            end
        end)

        local dragging = false
        local dragInput, dragStart, startPos
        local function update(input)
            local delta = input.Position - dragStart
            MIV.BHI.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
        MIV.BHI.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = MIV.BHI.Position
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                    end
                end)
            end
        end)
        MIV.BHI.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                dragInput = input
            end
        end)
        UserInput.InputChanged:Connect(function(input)
            if input == dragInput and dragging then
                update(input)
            end
        end)
    end

    function InventoryViewer.start()
        if running then return end
        running = true

        local ok = buildGui()
        if not ok then
            running = false
            return
        end

        hookGui()
    end

    function InventoryViewer.stop()
        if not running then return end
        running = false

        disconnectAll()

        if MIV.SG then
            MIV.SG:Destroy()
        end
        MIV = {}
        SOT = nil
    end
end

local CONFIG = "ProjectAstra_config.json"
local canIO = typeof(writefile) == "function"
    and typeof(readfile) == "function"
    and typeof(isfile) == "function"

local function packColor(c)
    return { c.R, c.G, c.B }
end

local function unpackColor(t, fallback)
    if type(t) == "table" and #t == 3 then
        return Color3.new(t[1], t[2], t[3])
    end
    return fallback
end

local function saveConfig()
    if not canIO then return end
    local headers = {}
    for i, c in ipairs(state.headers) do
        headers[i] = packColor(c)
    end
    local payload = {
        transparency = state.transparency,
        color        = packColor(state.color),
        gradient     = state.gradient,
        autosave     = state.autosave,
        headers      = headers,
        feature      = packColor(state.featureColor),
    }
    pcall(function()
        writefile(CONFIG, Http:JSONEncode(payload))
    end)
end

local function loadConfig()
    if not canIO or not isfile(CONFIG) then return end
    local ok, data = pcall(function()
        return Http:JSONDecode(readfile(CONFIG))
    end)
    if not ok or type(data) ~= "table" or not data.autosave then return end
    state.autosave     = true
    state.transparency = tonumber(data.transparency) or state.transparency
    state.color        = unpackColor(data.color, state.color)
    if type(data.gradient) == "boolean" then
        state.gradient = data.gradient
    end
    state.featureColor = unpackColor(data.feature, state.featureColor)
    if type(data.headers) == "table" then
        for i = 1, #CATEGORIES do
            state.headers[i] = unpackColor(data.headers[i], state.headers[i])
        end
    end
end

loadConfig()
draft = shallowCopy(state)
draft.headers = shallowCopy(state.headers)

local function instance(class, parent, props)
    local obj = Instance.new(class)
    for k, v in pairs(props) do
        obj[k] = v
    end
    if parent then obj.Parent = parent end
    return obj
end

local function round(obj, r)
    return instance("UICorner", obj, { CornerRadius = r or UDim.new(0, 8) })
end

local function outline(obj, color, thickness, alpha)
    return instance("UIStroke", obj, {
        Color = color or C.accent,
        Thickness = thickness or 1,
        Transparency = alpha or 0.5,
    })
end

local function fill(obj, stops, rotation)
    return instance("UIGradient", obj, {
        Color = ColorSequence.new(stops),
        Rotation = rotation or 90,
    })
end

local function fill2(obj, a, b, rotation)
    return fill(obj, {
        ColorSequenceKeypoint.new(0, a),
        ColorSequenceKeypoint.new(1, b),
    }, rotation)
end

end

local function isMove(input)

end

local hue, sat, val = 0, 1, 1
local palColor = draft.color

local activeDrag = nil

local function toHex(c)
    return string.format("#%02X%02X%02X",
        math.floor(c.R * 255 + 0.5),
        math.floor(c.G * 255 + 0.5),
        math.floor(c.B * 255 + 0.5))
end

local function refreshPalette()
    palColor = Color3.fromHSV(hue, sat, val)
    svArea.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
    svCursor.Position = UDim2.new(sat, 0, 1 - val, 0)
    hueCursor.Position = UDim2.new(hue, 0, 0.5, 0)
    preview.BackgroundColor3 = palColor
    hexLabel.Text = toHex(palColor)
end

local function setPalette(c)
    hue, sat, val = c:ToHSV()
    refreshPalette()
end

local function openPalette(target, color)
    palTarget = target
    setPalette(color)
    palette.Visible = true
end

local function normFromMouse(obj)
    local mp = UserInput:GetMouseLocation()
    local ap, as = obj.AbsolutePosition, obj.AbsoluteSize
    return math.clamp((mp.X - ap.X) / as.X, 0, 1),
           math.clamp((mp.Y - ap.Y) / as.Y, 0, 1)
end

local function applyDrag()
    if activeDrag == "sv" then
        local x, y = normFromMouse(svArea)
        sat, val = x, 1 - y
    elseif activeDrag == "hue" then
        hue = normFromMouse(hueBar)
    end
    refreshPalette()
end

local colFrames = {}
local headerRefs = {}
local rowRefs = {}
local allRows = {}

local function applyHeaders()
    for i, ref in ipairs(headerRefs) do
        local base = state.headers[i]
        local dark = base:Lerp(C.black, 0.6)
        ref.grad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, base),
            ColorSequenceKeypoint.new(1, dark),
        })
        ref.patch.BackgroundColor3 = dark
    end
end

local function applyFeatureColor()
    local base = state.featureColor
    local dark = base:Lerp(C.black, 0.4)
    for _, ref in ipairs(rowRefs) do
        ref.grad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, base),
            ColorSequenceKeypoint.new(1, dark),
        })
    end
end

local function applyStyle()
    local c0 = state.color:Lerp(C.accent, 0.18)
    local c1 = state.color:Lerp(C.black, 0.4)
    for _, ref in ipairs(colFrames) do
        ref.frame.BackgroundTransparency = state.transparency
        ref.frame.BackgroundColor3 = state.gradient and C.white or state.color
        ref.grad.Enabled = state.gradient
        ref.grad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, c0),
            ColorSequenceKeypoint.new(0.5, state.color),
            ColorSequenceKeypoint.new(1, c1),
        })
    end
    applyHeaders()
    applyFeatureColor()
end

local function pulseRow(row, nameLbl)
    Tween:Create(row, TweenInfo.new(0.08, Enum.EasingStyle.Quad), {
        BackgroundTransparency = 0.35,
    }):Play()
    nameLbl.TextColor3 = C.text
    task.delay(0.5, function()
        Tween:Create(row, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
            BackgroundTransparency = 1,
        }):Play()
        nameLbl.TextColor3 = C.textOff
    end)
end

local function runHandler(name, on)
    local handler = HANDLERS[name]
    if not handler then return end
    if on then
        if handler.start then handler.start() end
    else
        if handler.stop then handler.stop() end
    end
end

local function buildColumn(data)
    local col = instance("Frame", holder, {
        Name = "Col_" .. data.title,
        Size = UDim2.new(IS_MOBILE and 0.45 or 0.23, 0, 1, 0),
        BackgroundColor3 = state.color,
        BackgroundTransparency = state.transparency,
        BorderSizePixel = 0,
    })
    round(col, UDim.new(0, 10))
    local colGrad = fill2(col, C.accent, C.black, 160)
    table.insert(colFrames, { frame = col, grad = colGrad })

    local colHeader = instance("Frame", col, {
        Size = UDim2.new(1, 0, 0, 38),
        BackgroundColor3 = C.white,
        BackgroundTransparency = 0.1,
        BorderSizePixel = 0,
        ZIndex = 2,
    })
    round(colHeader, UDim.new(0, 10))
    local patch = instance("Frame", colHeader, {
        Size = UDim2.new(1, 0, 0, 10),
        Position = UDim2.new(0, 0, 1, -10),
        BackgroundColor3 = C.header,
        BorderSizePixel = 0,
        ZIndex = 2,
    })
    local headerGrad = fill2(colHeader, C.header, C.header:Lerp(C.black, 0.6), 180)
    table.insert(headerRefs, { header = colHeader, patch = patch, grad = headerGrad })

    instance("TextLabel", colHeader, {
        Size = UDim2.new(1, -12, 1, 0),
        Position = UDim2.new(0, 10, 0, 0),
        BackgroundTransparency = 1,
        Text = data.title,
        TextColor3 = C.title,
        TextSize = 13,
        Font = Enum.Font.GothamBold,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 3,
    })

    local scroll = instance("ScrollingFrame", col, {
        Size = UDim2.new(1, 0, 1, -42),
        Position = UDim2.new(0, 0, 0, 42),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ScrollBarThickness = IS_MOBILE and 4 or 2,
        ScrollBarImageColor3 = Color3.fromRGB(90, 80, 180),
        CanvasSize = UDim2.new(0, 0, 0, #data.items * (IS_MOBILE and 38 or 32) + 8),
        ScrollingDirection = Enum.ScrollingDirection.Y,
    })
    instance("UIListLayout", scroll, { Padding = UDim.new(0, 2) })
    instance("UIPadding", scroll, {
        PaddingLeft  = UDim.new(0, 5),
        PaddingRight = UDim.new(0, 5),
        PaddingTop   = UDim.new(0, 4),
    })

    for _, item in ipairs(data.items) do
        local on = false
        local pulse = item.pulse == true

        local rowHeight = IS_MOBILE and 34 or 28
        local row = instance("Frame", scroll, {
            Name = item.name,
            Size = UDim2.new(1, 0, 0, rowHeight),
            BackgroundColor3 = C.white,
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
        })
        round(row, UDim.new(0, 6))
        local rowGrad = fill2(row, C.accent, C.accent:Lerp(C.black, 0.4), 180)
        table.insert(rowRefs, { frame = row, grad = rowGrad })

        local nameSize
        if pulse then
            nameSize = UDim2.new(1, -12, 1, 0)
        else
            nameSize = UDim2.new(1, -44, 1, 0)
        end

        local nameLbl = instance("TextLabel", row, {
            Size = nameSize,
            Position = UDim2.new(0, 6, 0, 0),
            BackgroundTransparency = 1,
            Text = item.name,
            TextColor3 = C.textOff,
            TextSize = IS_MOBILE and 12 or 11,
            Font = Enum.Font.Gotham,
            TextXAlignment = Enum.TextXAlignment.Left,
        })

        local rowToggle
        if not pulse then
            rowToggle = buildRowToggle(row)
            paintRowToggle(rowToggle, on, false)
        end

        local hit = instance("TextButton", row, {
            Size = UDim2.new(1, 0, 1, 0),
            BackgroundTransparency = 1,
            Text = "",
        })

        local function activate()
            if pulse then
                pulseRow(row, nameLbl)
                runHandler(item.name, true)
                return
            end
            on = not on
            Tween:Create(row, TweenInfo.new(0.15, Enum.EasingStyle.Quad), {
                BackgroundTransparency = on and 0.65 or 1,
            }):Play()
            nameLbl.TextColor3 = on and C.text or C.textOff
            paintRowToggle(rowToggle, on, true)
            runHandler(item.name, on)
        end

        hit.MouseButton1Click:Connect(activate)
        hit.TouchTap:Connect(activate)

        table.insert(allRows, { frame = row, name = item.name:lower() })
    end
end

    local function beginDrag(input)

            dragging, moved = true, false

        if isPress(input) and dragging then
            dragging = false
            if not moved then
                toggleMenu()
            end
        end
    end

    logoBtn.InputBegan:Connect(beginDrag)
    logoBtn.InputEnded:Connect(endDrag)

    UserInput.InputChanged:Connect(function(input)
        if dragging and isMove(input) then
            local delta = UserInput:GetMouseLocation() - startMouse
            local threshold = IS_MOBILE and 10 or 4
            if delta.Magnitude > threshold then moved = true end
            if moved then
                logoBtn.Position = UDim2.fromOffset(
                    startPos.X + delta.X,
                    startPos.Y + delta.Y
                )
            end
        end
    end)
end

searchBox:GetPropertyChangedSignal("Text"):Connect(function()
    local q = searchBox.Text:lower()
    for _, entry in ipairs(allRows) do
        if q == "" or entry.name:find(q, 1, true) ~= nil then
            entry.frame.Visible = true
        else
            entry.frame.Visible = false
        end
    end
end)

        refreshSettings()
        settings.Visible = true
    end
end

local function applyPalette()
    if palTarget == "header" then
        if catIndex == 0 then
            for i = 1, #CATEGORIES do
                draft.headers[i] = palColor
            end
        else
            draft.headers[catIndex] = palColor
        end
    elseif palTarget == "feature" then
        draft.featureColor = palColor
    else
        draft.color = palColor
    end
    refreshSettings()
    palette.Visible = false
    activeDrag = nil
end

local resetArmed = false
local resetToken = 0

local function disarmReset()
    resetArmed = false
    resetBtn.Text = "Reset"
    resetBtn.TextColor3 = C.textOff
    resetBtn.BackgroundColor3 = C.field
end

local function onReset()
    if not resetArmed then
        resetArmed = true
        resetToken = resetToken + 1
        local myToken = resetToken
        resetBtn.Text = "Sure?"
        resetBtn.TextColor3 = C.text
        resetBtn.BackgroundColor3 = Color3.fromRGB(150, 50, 60)
        task.delay(3, function()
            if resetArmed and resetToken == myToken then
                disarmReset()
            end
        end)
        return
    end

    disarmReset()
    state = defaults()
    draft = shallowCopy(state)
    draft.headers = shallowCopy(state.headers)
    palette.Visible = false
    activeDrag = nil
    applyStyle()
    refreshSettings()
    saveConfig()
end

resetBtn.MouseButton1Click:Connect(onReset)
resetBtn.TouchTap:Connect(onReset)

UserInput.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.Insert then
        toggleMenu()
    end
end)
